    if(!window[VALNET_MAP_NAMESPACE]) {
        window[VALNET_MAP_NAMESPACE] = {
            isListeningForLoadEvents: false,
            mapIdsToLoad: [],
            getUserState: (idMap) => {
                const markers = window[idMap].markers.reduce((acc, marker) => {
                    if(marker.status === 'complete') {
                        acc[marker.uid] = { status: marker.status };
                    }
                    return acc;
                }, {});
                return { markers };
            },
            saveUserState: (idMap) => {
                window.dispatchEvent(new CustomEvent("vn_user_map:save", {
                    detail: { map_id: idMap, state: window[VALNET_MAP_NAMESPACE].getUserState(idMap) }
                }));
            },
            loadUserState: (idMap, state) => {
                if(!state) {
                    return;
                }
                if(typeof state.markers !== 'object') {
                    return;
                }
                const markers = window[idMap].markers.map(marker => {
                    if(state.markers[marker.uid]) {
                        marker = {
                            ...marker,
                            ...state.markers[marker.uid]
                        };
                    }
                    return marker;
                });
                window[idMap].setView({ markers });
            },
            initUserState: (idMap) => {
                if(window[VALNET_MAP_NAMESPACE].mapIdsToLoad.includes(idMap)) return;
                window[VALNET_MAP_NAMESPACE].mapIdsToLoad.push(idMap);
                window.dispatchEvent(new CustomEvent("vn_user_map:init", {
                    detail: { map_id: idMap }
                }));
            }
        };
    }
    if(!window[VALNET_MAP_NAMESPACE].isListeningForLoadEvents) {
        window.addEventListener("vn_user_map:load", (event) => {
            if(!event.detail) return;
            if(!window[VALNET_MAP_NAMESPACE].mapIdsToLoad.includes(event.detail.map_id)) return;
            window[VALNET_MAP_NAMESPACE].loadUserState(event.detail.map_id, event.detail.state);
        });
